# ============================================================================
#  TFAST Fun Pack -- character traits                          CK3 1.20.x
# ============================================================================
#  Reputation traits earned through the court-life decisions.
#
#  category = fame matches how vanilla models standing traits (drunkard,
#  reveler, and so on). Each icon reuses an EXISTING vanilla texture from
#  gfx/interface/icons/traits/, so the mod ships no art and cannot conflict.
# ============================================================================


# Won the crown at a great tournament.
tfast_fun_tournament_champion = {
	category = fame
	icon = blademaster.dds

	martial = 2
	prowess = 3

	same_opinion = 10
	opposite_opinion = -5

	ruler_designer_cost = 20

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { NOT = { exists = this } }
				desc = trait_tfast_fun_tournament_champion_desc
			}
			desc = trait_tfast_fun_tournament_champion_character_desc
		}
	}
}
