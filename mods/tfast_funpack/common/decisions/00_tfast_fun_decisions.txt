# ============================================================================
#  TFAST Fun Pack -- court life & character                CK3 1.20.x
# ============================================================================
#  Things to do that are about the ruler's court and personality rather than
#  about conquest or administration. Every one costs something real and leaves
#  a lasting mark, so they read as memorable choices rather than buttons.
#
#  SAFETY CONTRACT (same as every TFAST mod)
#    is_ai = no  +  ai_check_interval = 0   -> only the human player can see or
#    take these, so the AI's economy and wars are never affected.
#    Additive only: no vanilla file is overwritten.
#
#  Every effect, trigger, modifier key, trait and illustration below was
#  verified against the installed 1.20.0.3 data.
# ============================================================================


# ---------------------------------------------------------------------------
# 1. A great tournament, with ransoms, melees and a champion's crown.
# ---------------------------------------------------------------------------
tfast_fun_grand_tournament = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_knight_kneeling.dds" }
	decision_group_type = major
	sort_order = 9300
	ai_check_interval = 0

	is_shown = { is_ai = no  is_landed = yes  gold >= 400 }
	is_valid = { is_landed = yes }

	cost = { gold = 400 }
	cooldown = { days = 1825 }

	effect = {
		custom_tooltip = tfast_fun_grand_tournament_tt
		add_trait = tfast_fun_tournament_champion
		add_prestige = 400
		add_character_modifier = {
			modifier = tfast_fun_tournament_modifier
			years = 20
		}
	}
}
