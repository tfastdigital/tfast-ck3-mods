# ============================================================================
#  TFAST History -- historic acts & institutions                 CK3 1.20.x
# ============================================================================
#  Ten decisions that let you enact things that really happened in the
#  medieval world, with real costs and cooldowns so they read as history
#  rather than as cheats.
#
#  SAFETY CONTRACT (identical to the rest of the TFAST suite)
#  ---------------------------------------------------------
#   * is_ai = no             -> only the human player can ever take these, so
#                               the AI's economy and wars are untouched.
#   * ai_check_interval = 0  -> the AI never even evaluates them.
#   * Additive only          -> no vanilla file is overwritten.
#   * Every effect, trigger, modifier key and illustration below was verified
#     against the installed 1.20.0.3 data.
#
#  Each decision grants a period of standing rather than instant power, and
#  most also confer the matching reputation trait.
# ============================================================================


# ---------------------------------------------------------------------------
# 1. A merchant league of the northern seas.
#    Real parallel: the Hanseatic League, from the 12th century onward.
# ---------------------------------------------------------------------------
tfast_hist_hanseatic_league = {
	picture = { reference = "gfx/interface/illustrations/decisions/tgp_silk_road.dds" }
	decision_group_type = major
	sort_order = 9200
	ai_check_interval = 0

	is_shown = {
		is_ai = no
		is_landed = yes
		highest_held_title_tier >= tier_duchy
		capital_county = { is_coastal_county = yes }
	}
	is_valid = { is_landed = yes }

	cost = { gold = 500 }
	cooldown = { days = 1825 }

	effect = {
		custom_tooltip = tfast_hist_hanseatic_league_tt
		add_trait = tfast_hist_league_founder
		add_character_modifier = {
			modifier = tfast_league_charter_modifier
			years = 25
		}
		add_prestige = 500
		trigger_event = tfast_history.0001
	}
}
