# ============================================================================
#  TFAST Africa -- historical decisions                     CK3 1.20.x
# ============================================================================
#  Twelve decisions drawn from documented African history. Every one is a real
#  institution, practice or event, named as a generic to stay playable across
#  cultures and dates.
#
#  SAFETY CONTRACT (identical to every TFAST mod)
#    is_ai = no  +  ai_check_interval = 0  -> only the human player ever sees
#    these, so the AI's economy, wars and balance are untouched.
#    Additive only: no vanilla file is overwritten.
#    Every effect, trigger, modifier key and illustration verified against
#    1.20.0.3.
# ============================================================================


# ---------------------------------------------------------------------------
# 1. Open the Trans-Saharan Gold Road
#    Mali and Ghana grew rich moving Wangara gold north against Saharan salt.
# ---------------------------------------------------------------------------
tfast_africa_gold_road = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_golden_age.dds" }
	decision_group_type = major
	sort_order = 9400
	ai_check_interval = 0

	is_shown = {
		is_ai = no
		is_landed = yes
		tfast_africa_is_west_african_trigger = yes
	}
	is_valid = { is_landed = yes }

	cost = { gold = 300 }
	cooldown = { days = 3650 }

	effect = {
		custom_tooltip = tfast_africa_gold_road_tt
		add_trait = tfast_africa_gold_merchant
		add_character_modifier = {
			modifier = tfast_africa_gold_road_modifier
			years = 25
		}
		add_prestige = 300
	}
}
