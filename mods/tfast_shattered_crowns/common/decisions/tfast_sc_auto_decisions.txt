# ============================================================================
#  TFAST Shattered Crowns -- the automation menu               CK3 1.20.x
# ============================================================================
#  One menu, seven switches. Open it from the Decisions tab, flip what you
#  want the realm to handle by itself, close it again:
#
#    Auto: Maintain the Wall      fund upkeep / patch the stones
#    Auto: Reinforce the Wall     refit when there is coin to spare
#    Auto: Rebuild Counties       clear the raided modifier
#    Auto: Tell the Tale          use a waiting memory
#    Auto: Rise Against the Hollow  during the Long Dark
#    Auto: Break Blockades        pay the escorts
#    Auto: Hire Mercenaries       while at war, keeping a reserve
#
#  Every switch is a character flag (tfast_sc_auto_*). The yearly pass
#  (tfast_sc_automation_yearly_effect) only ever does what the matching
#  decision could have done, at the same price. is_ai = no throughout:
#  the AI never sees the menu and never automates anything.
# ============================================================================


# ---------------------------------------------------------------------------
#  OPEN / CLOSE
# ---------------------------------------------------------------------------
tfast_sc_auto_menu_open = {
	picture = { reference = "gfx/interface/illustrations/decisions/tfast_sc_hollow_omens_decision.dds" }
	decision_group_type = major
	sort_order = 9300
	ai_check_interval = 0

	is_shown = {
		is_ai = no
		NOT = { has_character_flag = tfast_sc_auto_menu }
	}
	is_valid = { is_ai = no }

	effect = { add_character_flag = { flag = tfast_sc_auto_menu } }
}

tfast_sc_auto_menu_close = {
	picture = { reference = "gfx/interface/illustrations/decisions/tfast_sc_reckoning_decision.dds" }
	decision_group_type = major
	sort_order = 9301
	ai_check_interval = 0

	is_shown = {
		is_ai = no
		has_character_flag = tfast_sc_auto_menu
	}
	is_valid = { is_ai = no }

	effect = { remove_character_flag = tfast_sc_auto_menu }
}
