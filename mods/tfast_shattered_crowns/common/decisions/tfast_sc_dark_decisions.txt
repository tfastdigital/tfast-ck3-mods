# ---------------------------------------------------------------------------
#  REINFORCE THE WALL
#  Two levels of refacing for a standing wall. Nothing beyond stage gates.
# ---------------------------------------------------------------------------
tfast_sc_wall_reinforce = {
	picture = { reference = "gfx/interface/illustrations/decisions/tfast_sc_winter_wall_decision.dds" }
	decision_group_type = major
	sort_order = 9523
	ai_check_interval = 0

	is_shown = {
		is_ai = no
		is_landed = yes
		has_variable = tfast_sc_wall_stage
		var:tfast_sc_wall_stage >= 3
		OR = {
			NOT = { has_variable = tfast_sc_wall_upgrade_level }
			var:tfast_sc_wall_upgrade_level < 2
		}
	}
	is_valid = {
		has_variable = tfast_sc_wall_stage
		var:tfast_sc_wall_stage >= 3
	}

	cost = { gold = 300 }

	effect = {
		custom_tooltip = tfast_sc_wall_reinforce_tt
		tfast_sc_wall_reinforce_effect = yes
	}
}
