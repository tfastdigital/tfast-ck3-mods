# ============================================================================
#  TFAST Shattered Crowns -- decisions                        CK3 1.20.x
# ============================================================================
#  Same safety contract as every TFAST mod: is_ai = no + ai_check_interval = 0
#  for every player-facing decision. Additive only. Every effect, trigger and
#  illustration verified against the installed 1.20 data.
#
#  The Hidden Chosen carries NO decision until awakening. The True Death
#  decision is visible only to an awakened chosen who is the human player,
#  only while at war with the Hollow, and never to an ordinary mortal.
# ============================================================================


##############################################################
# THE HOLLOW TIDE (Situations window)
##############################################################
# History starts this for a new campaign. An already-open save does not
# replay history, so this decision puts the tide in the Situations window
# the moment it is clicked. It hides once this character is a participant.
tfast_sc_open_the_tide = {
	picture = { reference = "gfx/interface/illustrations/decisions/tfast_sc_hollow_omens_decision.dds" }
	decision_group_type = major
	sort_order = 9390
	ai_check_interval = 0

	is_shown = {
		is_ai = no
		is_landed = yes
		NOT = {
			any_character_situation = {
				situation_type = tfast_sc_hollow_tide
			}
		}
	}
	is_valid = { is_landed = yes }

	effect = {
		custom_tooltip = tfast_sc_open_the_tide_tt
		tfast_sc_ensure_tide_effect = yes
	}
}


##############################################################
# FALLBACK BOOTSTRAP (repair path)
##############################################################
# If the campaign for any reason started without the Hollow Dominion being
# raised (a failed game-start hook, an exotic load order, a converted save),
# this decision appears to the human player once, and raises it on click.
# It disappears forever once the Dominion exists. It is the ultimate safety
# net for the whole mod - no on_action hook is required for it to work.
tfast_sc_raise_the_hollow = {
	picture = { reference = "gfx/interface/illustrations/decisions/tfast_sc_hollow_omens_decision.dds" }
	decision_group_type = major
	sort_order = 9400
	ai_check_interval = 0

	is_shown = {
		is_ai = no
		NOT = { tfast_sc_hollow_initialized_trigger = yes }
	}
	is_valid = {
		NOT = { tfast_sc_hollow_initialized_trigger = yes }
	}

	effect = {
		custom_tooltip = tfast_sc_raise_the_hollow_tt
		tfast_sc_bootstrap_effect = yes
		tfast_sc_capture_designated_dynasty_effect = yes
		tfast_sc_ensure_tide_effect = yes
	}
}

# ---------------------------------------------------------------------------
#  THE WORLD COUNCIL PLEDGE
#  Tied to the crisis: only offered once the Reckoning has been foretold.
#  This is not an alliance with the Hollow - it is an oath against it.
# ---------------------------------------------------------------------------
tfast_sc_world_council_pledge = {
	picture = { reference = "gfx/interface/illustrations/decisions/tfast_sc_reckoning_decision.dds" }
	decision_group_type = major
	sort_order = 9512
	ai_check_interval = 0

	is_shown = {
		is_ai = no
		is_landed = yes
		exists = global_var:tfast_sc_crisis_phase
		global_var:tfast_sc_crisis_phase >= 2
		NOT = { has_variable = tfast_sc_world_pledge }
	}
	is_valid = { is_landed = yes }

	cost = { prestige = 200 }
	cooldown = { days = 3650 }

	effect = {
		custom_tooltip = tfast_sc_world_council_tt
		set_variable = { name = tfast_sc_world_pledge value = yes years = 25 }
		add_character_modifier = {
			modifier = tfast_sc_world_pledge_modifier
			years = 25
		}
		if = {
			limit = { exists = global_var:tfast_sc_world_resistance }
			change_global_variable = { name = tfast_sc_world_resistance add = 1 }
		}
		else = {
			set_global_variable = { name = tfast_sc_world_resistance value = 1 }
		}
	}
}
