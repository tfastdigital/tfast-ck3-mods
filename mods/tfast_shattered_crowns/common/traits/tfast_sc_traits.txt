# ============================================================================
#  TFAST Shattered Crowns -- traits                           CK3 1.20.x
# ============================================================================
#  Verified against game/common/traits/_traits.info and 00_traits.txt:
#    * `immortal = yes` stops visual aging and natural death, and is exactly
#      the engine flag vanilla's own `immortal` trait uses (line 17510).
#    * `shown_in_ruler_designer = no` keeps these out of the ruler designer.
#    * `flag = x` values are referenced by script via `has_trait_flag`, and
#      the game localizes them automatically from TRAIT_FLAG_DESC_<flag>.
#  Icons are original tfast_sc_*.dds shields in gfx/interface/icons/traits/.
#  Format matches vanilla immortal.dds: 120x120, uncompressed 32-bit BGRA.
# ============================================================================


# ---------------------------------------------------------------------------
#  The Hollow are not alive in the way the living are.
#  Granted by the bootstrap to the Overlord and all of his blood.
#  Immortal, unaging, but script-killable (True Death only).
# ---------------------------------------------------------------------------
tfast_sc_hollow_immortal = {
	icon = "tfast_sc_icon_immortal.dds"
	shown_in_ruler_designer = no

	opposites = { incapable }

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { NOT = { exists = this } }
				desc = trait_tfast_sc_hollow_immortal_desc
			}
			desc = trait_tfast_sc_hollow_immortal_character_desc
		}
	}

	immortal = yes
	flag = tfast_sc_hollow_immortal
	no_prowess_loss_from_age = yes
	long_reign_bonus_mult = -1000

	# The Hollow do not sicken.
	health = 10
	fertility = 0.25
}
