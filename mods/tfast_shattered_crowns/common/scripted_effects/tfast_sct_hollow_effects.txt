# The Hollow hunt the blood of Darkness's killer: about once a decade the Overlord marches on a Targaryen ruler.
tfast_sct_hollow_vengeance_effect = {
	if = {
		limit = {
			tfast_sc_overlord_valid_trigger = yes
			global_var:tfast_sc_overlord = {
				NOT = { has_character_flag = tfast_sc_overlord_sealed }
				NOT = { has_character_flag = tfast_sct_vengeance_cd }
			}
			global_var:tfast_sc_hollow_readiness >= 50
		}
		random_living_character = {
			limit = {
				has_character_flag = tfast_sc_targaryen
				is_landed = yes
				NOT = { has_character_flag = tfast_sc_hollow_host }
				NOT = { is_at_war_with = global_var:tfast_sc_overlord }
			}
			save_temporary_scope_as = tfast_sct_vengeance_target
		}
		if = {
			limit = { exists = scope:tfast_sct_vengeance_target }
			global_var:tfast_sc_overlord = {
				add_character_flag = { flag = tfast_sct_vengeance_cd years = 10 }
			}
			tfast_sc_declare_reckoning_war_effect = { TARGET = scope:tfast_sct_vengeance_target }
			scope:tfast_sct_vengeance_target = {
				if = {
					limit = { is_ai = no }
					trigger_event = tfast_sct.0040
				}
			}
			debug_log = "TFAST Shattered Crowns: the Hollow march on a Targaryen"
		}
	}
}
