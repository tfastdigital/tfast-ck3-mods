# ============================================================================
#  TFAST Shattered Crowns -- the Great Wall project           CK3 1.20.x
# ============================================================================
#  One wall project per sponsor, spanning a bounded set of EXISTING counties
#  (the sponsor's capital county and their other counties in the same de jure
#  duchy, capped at four). Stages, driven by decisions:
#
#    1 planning   --  free; designates the wall counties and the gate
#    2 towers     --  gold;   +fort_level, +levies
#    3 rampart    --  gold;   +fort_level, +defender advantage
#    4 gates      --  gold;   the gate county closes: SCRIPTED Hollow raids
#                     are redirected into tfast_sc_wall_defense_effect and
#                     cannot take gold until the gate is breached
#    5 wards      --  piety;  the Hollow's magic breaks against the wards
#
#  ENGINE LIMIT, STATED PLAINLY: this wall cannot change native CK3 army
#  pathfinding and cannot stop vanilla wars. It blocks the Hollow's scripted
#  raid director (the only Hollow raiding path generated by this mod) and it
#  slows real sieges through fort-level county modifiers. Nothing here
#  claims otherwise.
#
#  Continuity: everything lives in variables and county modifiers, so the
#  wall survives ruler death and succession unharmed (see upkeep below).
# ============================================================================


# ---------------------------------------------------------------------------
#  PLANNING
#  Root = the sponsor. Called by the planning decision.
# ---------------------------------------------------------------------------
tfast_sc_wall_plan_effect = {
	save_scope_as = tfast_sc_wall_sponsor_scope
	set_variable = { name = tfast_sc_wall_stage value = 1 }
	set_variable = { name = tfast_sc_wall_integrity value = 100 }
	set_variable = { name = tfast_sc_wall_upkeep_years value = 25 }
	clear_variable_list = tfast_sc_wall_counties
	set_variable = { name = tfast_sc_wall_count_guard value = 0 }

	# The gate is the capital, always.
	capital_county = {
		set_variable = { name = tfast_sc_wall_county value = yes }
		set_variable = { name = tfast_sc_wall_sponsor value = scope:tfast_sc_wall_sponsor_scope }
		set_variable = { name = tfast_sc_wall_gate value = yes }
	}
	set_variable = { name = tfast_sc_wall_gate_county value = capital_county }
	add_to_variable_list = {
		name = tfast_sc_wall_counties
		target = capital_county
	}
	change_variable = { name = tfast_sc_wall_count_guard add = 1 }

	# The rest of the sponsor's own counties in the same de jure duchy
	# (capped at four in total).
	capital_county.duchy ?= {
		every_de_jure_county = {
			save_temporary_scope_as = tfast_sc_wall_county_scope
			if = {
				limit = {
					holder = { this = scope:tfast_sc_wall_sponsor_scope }
					scope:tfast_sc_wall_sponsor_scope = {
						NOT = {
							is_target_in_variable_list = {
								name = tfast_sc_wall_counties
								target = scope:tfast_sc_wall_county_scope
							}
						}
						var:tfast_sc_wall_count_guard < 4
					}
				}
				set_variable = { name = tfast_sc_wall_county value = yes }
				set_variable = { name = tfast_sc_wall_sponsor value = scope:tfast_sc_wall_sponsor_scope }
				scope:tfast_sc_wall_sponsor_scope = {
					add_to_variable_list = {
						name = tfast_sc_wall_counties
						target = scope:tfast_sc_wall_county_scope
					}
					change_variable = { name = tfast_sc_wall_count_guard add = 1 }
				}
			}
		}
	}

	remove_variable = tfast_sc_wall_count_guard

	if = {
		limit = { is_ai = no }
		send_interface_toast = {
			type = event_toast_effect_good
			title = tfast_sc_wall_planned_toast.t
			left_icon = scope:tfast_sc_wall_sponsor_scope
		}
	}

	debug_log = "TFAST Shattered Crowns: a Great Wall project has been planned"
	debug_log_scopes = yes
}


# Drop a wall that was only planned, never built. Called when a Great Project
# is cancelled before construction starts. A wall that already has towers is
# left standing so the later decisions can finish it.
tfast_sc_wall_abandon_plan_effect = {
	if = {
		limit = { var:tfast_sc_wall_stage ?= 1 }
		every_in_list = {
			variable = tfast_sc_wall_counties
			remove_variable = tfast_sc_wall_county
			remove_variable = tfast_sc_wall_sponsor
			remove_variable = tfast_sc_wall_gate
		}
		remove_variable = tfast_sc_wall_stage
		remove_variable = tfast_sc_wall_integrity
		remove_variable = tfast_sc_wall_upkeep_years
		remove_variable = tfast_sc_wall_gate_county
		clear_variable_list = tfast_sc_wall_counties
		debug_log = "TFAST Shattered Crowns: a planned Great Wall was abandoned"
	}
}


# ---------------------------------------------------------------------------
#  STAGE 2 -- TOWERS
# ---------------------------------------------------------------------------
tfast_sc_wall_build_towers_effect = {
	if = {
		limit = { var:tfast_sc_wall_stage ?= 1 }
		every_in_list = {
			variable = tfast_sc_wall_counties
			add_county_modifier = { modifier = tfast_sc_wall_towers_modifier }
		}
		set_variable = { name = tfast_sc_wall_stage value = 2 }
		debug_log = "TFAST Shattered Crowns: wall towers raised"
	}
}


# ---------------------------------------------------------------------------
#  STAGE 3 -- RAMPART
# ---------------------------------------------------------------------------
tfast_sc_wall_build_rampart_effect = {
	if = {
		limit = { var:tfast_sc_wall_stage ?= 2 }
		every_in_list = {
			variable = tfast_sc_wall_counties
			add_county_modifier = { modifier = tfast_sc_wall_rampart_modifier }
		}
		set_variable = { name = tfast_sc_wall_stage value = 3 }
		debug_log = "TFAST Shattered Crowns: wall rampart raised"
	}
}


# ---------------------------------------------------------------------------
#  STAGE 4 -- GATES (the wall closes; scripted raids are blocked here)
# ---------------------------------------------------------------------------
tfast_sc_wall_build_gates_effect = {
	if = {
		limit = { var:tfast_sc_wall_stage ?= 3 }
		every_in_list = {
			variable = tfast_sc_wall_counties
			if = {
				limit = { has_variable = tfast_sc_wall_gate }
				add_county_modifier = { modifier = tfast_sc_wall_gate_modifier }
			}
			else = {
				add_county_modifier = { modifier = tfast_sc_wall_rampart_modifier }
			}
		}
		set_variable = { name = tfast_sc_wall_stage value = 4 }
		debug_log = "TFAST Shattered Crowns: the Great Wall's gates have closed"
	}
}

# ---------------------------------------------------------------------------
#  WALL DEFENSE
#  Root = the raid victim. Called from tfast_sc_raid_resolve_effect when the
#  victim's capital is a walled county with an intact gate.
#
#  The raid breaks against the wall unless it outmatches the defense; every
#  attempt costs the wall integrity, and a wall at zero integrity is
#  breached: the gate modifier falls and scripted raids pass freely again.
# ---------------------------------------------------------------------------
tfast_sc_wall_defense_effect = {
	save_scope_as = tfast_sc_victim

	capital_county = { save_scope_as = tfast_sc_walled_gate }
	var:tfast_sc_wall_sponsor = { save_scope_as = tfast_sc_wall_sponsor }

	# If the Host has no head, no assault comes at all: the watch stands
	# down and the wall simply holds.
	if = {
		limit = { NOT = { tfast_sc_overlord_valid_trigger = yes } }
		custom_tooltip = tfast_sc_wall_held_tt
		scope:tfast_sc_wall_sponsor = {
			tfast_sc_record_memory_effect = { MEMORY = tfast_sc_memory_wall_defense YEARS = 100 }
			trigger_event = { id = tfast_sc.0301 days = { 2 8 } }
		}
	}
	else = {
		global_var:tfast_sc_overlord = { save_scope_as = tfast_sc_raider }

		scope:tfast_sc_raider = {
			set_variable = {
				name = tfast_sc_raid_power
				value = tfast_sc_raid_strength_value
			}
		}

		# All comparisons happen inside the sponsor's own scope, variable
		# against variable, which is the vanilla-safe comparison pattern.
		scope:tfast_sc_wall_sponsor = {
			set_variable = { name = tfast_sc_defense_power value = tfast_sc_wall_defense_value }
			set_variable = {
				name = tfast_sc_wall_def_threat
				value = scope:tfast_sc_raider.var:tfast_sc_raid_power
			}
			set_variable = {
				name = tfast_sc_wall_def_threat_60
				value = {
					value = scope:tfast_sc_raider.var:tfast_sc_raid_power
					multiply = 0.6
				}
			}
		}

		# --------------------------------------------------------------
		# The wall holds.
		# --------------------------------------------------------------
		if = {
			limit = {
				scope:tfast_sc_wall_sponsor = { var:tfast_sc_defense_power >= var:tfast_sc_wall_def_threat }
			}
			custom_tooltip = tfast_sc_wall_held_tt
			scope:tfast_sc_wall_sponsor = {
				change_variable = { name = tfast_sc_wall_integrity add = -5 }
				tfast_sc_record_memory_effect = { MEMORY = tfast_sc_memory_wall_defense YEARS = 100 }
				trigger_event = { id = tfast_sc.0301 days = { 2 8 } }
			}
			scope:tfast_sc_raider = {
				change_global_variable = { name = tfast_sc_hollow_readiness add = -8 }
				add_opinion = {
					modifier = tfast_sc_wall_slighted_opinion
					target = scope:tfast_sc_wall_sponsor
				}
			}
			# In the Long Dark, a wall that holds is a deed of resistance.
			if = {
				limit = { tfast_sc_dark_reign_active_trigger = yes }
				tfast_sc_add_dark_resistance_effect = { AMOUNT = 1 }
			}
			if = {
				limit = { exists = global_var:tfast_sc_memory_wall_defenses }
				change_global_variable = { name = tfast_sc_memory_wall_defenses add = 1 }
			}
			else = {
				set_global_variable = { name = tfast_sc_memory_wall_defenses value = 1 }
			}
		}
		# --------------------------------------------------------------
		# The wall buckles but holds this time.
		# --------------------------------------------------------------
		else_if = {
			limit = {
				scope:tfast_sc_wall_sponsor = { var:tfast_sc_defense_power >= var:tfast_sc_wall_def_threat_60 }
			}
			custom_tooltip = tfast_sc_wall_buckled_tt
			scope:tfast_sc_wall_sponsor = {
				change_variable = { name = tfast_sc_wall_integrity add = -30 }
				tfast_sc_record_memory_effect = { MEMORY = tfast_sc_memory_wall_defense YEARS = 100 }
				trigger_event = { id = tfast_sc.0301 days = { 2 8 } }
			}
			capital_county = {
				change_county_control = minor_county_control_loss
			}
		}
		# --------------------------------------------------------------
		# The wall is broken open. This is the breach.
		# --------------------------------------------------------------
		else = {
			custom_tooltip = tfast_sc_wall_breached_tt
			scope:tfast_sc_wall_sponsor = {
				set_variable = { name = tfast_sc_wall_integrity value = 0 }
				tfast_sc_record_memory_effect = { MEMORY = tfast_sc_memory_wall_breached YEARS = 100 }
			}
			capital_county = {
				remove_county_modifier = tfast_sc_wall_gate_modifier
				set_variable = { name = tfast_sc_wall_breached value = yes }
				change_county_control = major_county_control_loss
				add_county_modifier = {
					modifier = tfast_sc_raided_modifier
					years = 10
				}
			}
			# The victim bleeds as the raid finally takes its cut.
			set_variable = {
				name = tfast_sc_loot
				value = { value = gold multiply = 0.2 floor = yes }
			}
			remove_short_term_gold = var:tfast_sc_loot
			scope:tfast_sc_raider = {
				add_gold = scope:tfast_sc_victim.var:tfast_sc_loot
			}
			set_variable = { name = tfast_sc_raid_cooldown value = yes years = 4 }

			# The world is told the wall has fallen.
			every_player = {
				trigger_event = { id = tfast_sc.0302 days = { 5 40 } }
			}
			debug_log = "TFAST Shattered Crowns: a Great Wall gate has been breached"
			debug_log_scopes = yes
		}

		# Integrity at zero ends the wall project's protections even when
		# the gate still stands (structurally the wall is gone).
		scope:tfast_sc_wall_sponsor = {
			if = {
				limit = { var:tfast_sc_wall_integrity <= 0 }
				set_variable = { name = tfast_sc_wall_integrity value = 0 }
				set_variable = { name = tfast_sc_wall_breached value = yes }
			}
		}

		scope:tfast_sc_raider = {
			remove_variable = tfast_sc_raid_power
		}
		scope:tfast_sc_wall_sponsor = {
			remove_variable = tfast_sc_defense_power
			remove_variable = tfast_sc_wall_def_threat
			remove_variable = tfast_sc_wall_def_threat_60
		}
	}
}


# ---------------------------------------------------------------------------
#  YEARLY UPKEEP
#  Root = the human player (called from tfast_sc_player_yearly_effect).
#  One wall project is standard; if the sponsor holds several through war,
#  the capital project is the one maintained here.
# ---------------------------------------------------------------------------
tfast_sc_wall_yearly_upkeep_effect = {
	if = {
		limit = {
			has_variable = tfast_sc_wall_stage
			var:tfast_sc_wall_stage >= 2
		}

		if = {
			limit = { var:tfast_sc_wall_upkeep_years > 0 }
			change_variable = { name = tfast_sc_wall_upkeep_years add = -1 }
		}
		else = {
			# Unfunded: the stones weather.
			change_variable = { name = tfast_sc_wall_integrity add = -15 }
			if = {
				limit = { var:tfast_sc_wall_integrity < 0 }
				set_variable = { name = tfast_sc_wall_integrity value = 0 }
			}
			trigger_event = { id = tfast_sc.0303 days = { 5 30 } }
			if = {
				limit = { var:tfast_sc_wall_integrity <= 0 }
				capital_county = { remove_county_modifier = tfast_sc_wall_gate_modifier }
				set_variable = { name = tfast_sc_wall_breached value = yes }
			}
		}
	}
}


# ---------------------------------------------------------------------------
#  REPAIR
#  Root = the sponsor. Called by the repair decision (cost paid there).
# ---------------------------------------------------------------------------
tfast_sc_wall_repair_effect = {
	change_variable = { name = tfast_sc_wall_integrity add = 40 }
	if = { limit = { var:tfast_sc_wall_integrity > 100 } set_variable = { name = tfast_sc_wall_integrity value = 100 } }
	change_variable = { name = tfast_sc_wall_upkeep_years add = 10 }

	# A breached gate is rehung once the wall can hold itself again.
	if = {
		limit = {
			var:tfast_sc_wall_stage >= 4
			capital_county = { has_variable = tfast_sc_wall_breached }
		}
		capital_county = {
			remove_variable = tfast_sc_wall_breached
			if = {
				limit = { NOT = { has_county_modifier = tfast_sc_wall_gate_modifier } }
				add_county_modifier = { modifier = tfast_sc_wall_gate_modifier }
			}
		}
	}
	debug_log = "TFAST Shattered Crowns: the Great Wall has been repaired"
	debug_log_scopes = yes
}
