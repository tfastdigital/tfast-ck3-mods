tfast_sc_seal_hollow_children_effect = {
	if = {
		limit = {
			exists = character:tfast_sc_overlord
			character:tfast_sc_overlord = { is_alive = yes NOT = { has_trait = tfast_sc_white_walker } }
		}
		character:tfast_sc_overlord = { add_trait = tfast_sc_white_walker }
	}
	if = {
		limit = {
			exists = character:tfast_sc_vaelith
			character:tfast_sc_vaelith = { is_alive = yes NOT = { has_trait = tfast_sc_white_walker } }
		}
		character:tfast_sc_vaelith = { add_trait = tfast_sc_white_walker }
	}
	tfast_sc_seal_one_child_effect = { WHO = tfast_sc_vaelor }
	tfast_sc_seal_one_child_effect = { WHO = tfast_sc_morvath }
	tfast_sc_seal_one_child_effect = { WHO = tfast_sc_kaedryn }
	tfast_sc_seal_one_child_effect = { WHO = tfast_sc_draven }
	tfast_sc_seal_one_child_effect = { WHO = tfast_sc_tharion }
	tfast_sc_seal_one_child_effect = { WHO = tfast_sc_seralyth }
	tfast_sc_seal_one_child_effect = { WHO = tfast_sc_nyxara }
	tfast_sc_seal_one_child_effect = { WHO = tfast_sc_velmora }
	tfast_sc_seal_one_child_effect = { WHO = tfast_sc_eryndra }
	tfast_sc_seal_one_child_effect = { WHO = tfast_sc_malethra }
}

# Root is not the child. WHO is the history id.
tfast_sc_seal_one_child_effect = {
	if = {
		limit = {
			exists = character:$WHO$
			character:$WHO$ = { is_alive = yes }
		}
		character:$WHO$ = {
			if = { limit = { NOT = { has_trait = tfast_sc_white_walker } } add_trait = tfast_sc_white_walker }
			if = { limit = { NOT = { has_trait = tfast_sc_hollow_unwed } } add_trait = tfast_sc_hollow_unwed }
			if = { limit = { NOT = { has_character_flag = tfast_sc_original_child } } add_character_flag = tfast_sc_original_child }
			if = {
				limit = { any_child = { count >= 2 is_alive = yes } }
				if = { limit = { NOT = { has_trait = tfast_sc_hollow_two_born } } add_trait = tfast_sc_hollow_two_born }
			}
			else = {
				if = { limit = { has_trait = tfast_sc_hollow_two_born } remove_trait = tfast_sc_hollow_two_born }
			}
			if = {
				limit = { is_married = yes }
				primary_spouse = { save_scope_as = hollow_unwanted_spouse }
				divorce = scope:hollow_unwanted_spouse
			}
			random_relation = {
				type = lover
				limit = { NOT = { has_character_flag = tfast_sc_original_child } }
				save_scope_as = hollow_outside_lover
			}
			if = {
				limit = { exists = scope:hollow_outside_lover }
				remove_relation_lover = scope:hollow_outside_lover
			}
		}
	}
}
