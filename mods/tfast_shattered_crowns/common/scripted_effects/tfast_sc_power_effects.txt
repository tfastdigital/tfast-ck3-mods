# Parents are the stronger beings. The Ten are high and each one is different,
# and none of them reach the parents. Courtiers do not give birth.
# Empty baronies in the parents' realm, and in their vassals' land, become castles
# and the castle is raised to the top level.

tfast_sc_raise_parent_skills_effect = {
	tfast_sc_one_parent_skills_effect = { WHO = tfast_sc_overlord }
	tfast_sc_one_parent_skills_effect = { WHO = tfast_sc_vaelith }
}

tfast_sc_one_parent_skills_effect = {
	if = {
		limit = {
			exists = character:$WHO$
			character:$WHO$ = {
				is_alive = yes
				NOT = { has_character_flag = tfast_sc_parent_skills }
			}
		}
		character:$WHO$ = {
			if = { limit = { diplomacy < 100 } add_diplomacy_skill = 83 }
			if = { limit = { martial < 100 } add_martial_skill = 80 }
			if = { limit = { stewardship < 100 } add_stewardship_skill = 83 }
			if = { limit = { intrigue < 100 } add_intrigue_skill = 79 }
			if = { limit = { learning < 100 } add_learning_skill = 67 }
			if = { limit = { prowess < 100 } add_prowess_skill = 78 }
			add_character_flag = tfast_sc_parent_skills
		}
	}
}

# Courtiers of the Hollow do not give birth. Human vassals are left alone,
# so a conquered ruler still has a living court of his own.
tfast_sc_quiet_courts_effect = {
	tfast_sc_quiet_one_court_effect = { WHO = tfast_sc_overlord }
	tfast_sc_quiet_one_court_effect = { WHO = tfast_sc_vaelith }
	tfast_sc_quiet_one_court_effect = { WHO = tfast_sc_vaelor }
	tfast_sc_quiet_one_court_effect = { WHO = tfast_sc_morvath }
	tfast_sc_quiet_one_court_effect = { WHO = tfast_sc_kaedryn }
	tfast_sc_quiet_one_court_effect = { WHO = tfast_sc_draven }
	tfast_sc_quiet_one_court_effect = { WHO = tfast_sc_tharion }
	tfast_sc_quiet_one_court_effect = { WHO = tfast_sc_seralyth }
	tfast_sc_quiet_one_court_effect = { WHO = tfast_sc_nyxara }
	tfast_sc_quiet_one_court_effect = { WHO = tfast_sc_velmora }
	tfast_sc_quiet_one_court_effect = { WHO = tfast_sc_eryndra }
	tfast_sc_quiet_one_court_effect = { WHO = tfast_sc_malethra }
}

tfast_sc_quiet_one_court_effect = {
	if = {
		limit = {
			exists = character:$WHO$
			character:$WHO$ = { is_alive = yes }
		}
		character:$WHO$ = {
			every_courtier = {
				limit = {
					is_alive = yes
					NOT = { has_character_modifier = tfast_sc_undead_court_modifier }
					NOT = { has_trait = tfast_sc_hollow_immortal }
				}
				add_character_modifier = { modifier = tfast_sc_undead_court_modifier }
				if = { limit = { NOT = { has_trait = tfast_sc_white_walker } } add_trait = tfast_sc_white_walker }
			}
		}
	}
}

tfast_sc_upgrade_family_land_effect = {
	tfast_sc_upgrade_one_realm_effect = { WHO = tfast_sc_overlord }
	tfast_sc_upgrade_one_realm_effect = { WHO = tfast_sc_vaelith }
}

tfast_sc_upgrade_one_realm_effect = {
	if = {
		limit = {
			exists = character:$WHO$
			character:$WHO$ = { is_alive = yes }
		}
		character:$WHO$ = {
			every_held_title = {
				limit = { tier = tier_barony }
				title_province = { tfast_sc_raise_permanent_castle_effect = yes }
			}
			every_vassal = {
				limit = { is_alive = yes }
				every_held_title = {
					limit = { tier = tier_barony }
					title_province = { tfast_sc_raise_permanent_castle_effect = yes }
				}
				every_vassal = {
					limit = { is_alive = yes }
					every_held_title = {
						limit = { tier = tier_barony }
						title_province = { tfast_sc_raise_permanent_castle_effect = yes }
					}
					every_vassal = {
						limit = { is_alive = yes }
						every_held_title = {
							limit = { tier = tier_barony }
							title_province = { tfast_sc_raise_permanent_castle_effect = yes }
						}
					}
				}
			}
		}
	}
}
