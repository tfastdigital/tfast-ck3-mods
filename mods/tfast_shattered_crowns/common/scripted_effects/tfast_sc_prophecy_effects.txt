# ============================================================================
#  TFAST Shattered Crowns -- the prophecy and the Chosen Emperor
#                                                            CK3 1.20.x
# ============================================================================
#  Design commitments (from the build spec, section 3.5):
#    * chosen only from the designated player dynasty, saved at campaign
#      initialization;
#    * after the first Great Reckoning only (the window opens in the
#      aftermath);
#    * either gender;
#    * hidden: a private variable, no trait, no public marker of any kind
#      until the awakening event;
#    * if the chosen dies, only a living DIRECT descendant may succeed -
#      never a cousin, never an unrelated house;
#    * if the bloodline is truly extinct, the prophecy ends explicitly.
# ============================================================================


# ---------------------------------------------------------------------------
#  MARK THE CHOSEN (hidden)
#  Root = the candidate child. Called by event tfast_sc.0400 after its
#  randomized delay. Re-checks eligibility so nothing leaks.
# ---------------------------------------------------------------------------
tfast_sc_prophecy_mark_effect = {
	if = {
		limit = {
			is_alive = yes
			tfast_sc_prophecy_open_trigger = yes
			NOT = { exists = global_var:tfast_sc_chosen_character }
			tfast_sc_in_designated_dynasty_trigger = yes
		}

		set_variable = { name = tfast_sc_chosen value = yes }
		add_character_flag = tfast_sc_chosen_hidden
		remove_variable = tfast_sc_prophecy_candidate
		set_global_variable = { name = tfast_sc_chosen_character value = this }
		set_global_variable = { name = tfast_sc_last_chosen value = this }

		# The prophecy story follows the chosen.
		create_story = {
			type = tfast_sc_prophecy
			save_scope_as = tfast_sc_prophecy_story
		}

		# The player is told only that something has begun - never what.
		every_player = {
			trigger_event = { id = tfast_sc.0402 days = { 20 60 } }
		}

		debug_log = "TFAST Shattered Crowns: a hidden chosen has been marked"
		debug_log_scopes = yes
	}
}


# ---------------------------------------------------------------------------
#  PROMOTE (succession inside the prophecy)
#  Root = the prophecy story. Param: CANDIDATE
# ---------------------------------------------------------------------------
tfast_sc_prophecy_promote_effect = {
	$CANDIDATE$ = {
		set_variable = { name = tfast_sc_chosen value = yes }
		add_character_flag = tfast_sc_chosen_hidden
	}
	set_global_variable = { name = tfast_sc_chosen_character value = $CANDIDATE$ }
	set_global_variable = { name = tfast_sc_last_chosen value = $CANDIDATE$ }
	make_story_owner = $CANDIDATE$
	every_player = {
		trigger_event = { id = tfast_sc.0402 days = { 20 60 } }
	}
	debug_log = "TFAST Shattered Crowns: the prophecy has passed to a direct descendant"
	debug_log_scopes = yes
}


# ---------------------------------------------------------------------------
#  SUCCESSION
#  Root = the prophecy story. Runs when the chosen died (story owner death)
#  and again on every prophecy check: children first, then grandchildren,
#  then great-grandchildren - and only the direct bloodline.
#  No cousin or unrelated branch is ever selected.
# ---------------------------------------------------------------------------
tfast_sc_prophecy_succession_effect = {
	# Already resolved?
	if = {
		limit = {
			exists = global_var:tfast_sc_chosen_character
			global_var:tfast_sc_chosen_character = { is_alive = yes }
		}
		# Keep the story pointed at the living chosen.
		if = {
			limit = { NOT = { story_owner = global_var:tfast_sc_chosen_character } }
			make_story_owner = global_var:tfast_sc_chosen_character
		}
	}
	else_if = {
		limit = {
			exists = global_var:tfast_sc_last_chosen
			global_var:tfast_sc_last_chosen = {
				any_child = {
					is_alive = yes
					tfast_sc_in_designated_dynasty_trigger = yes
				}
			}
		}
		global_var:tfast_sc_last_chosen = {
			ordered_child = {
				limit = {
					is_alive = yes
					tfast_sc_in_designated_dynasty_trigger = yes
				}
				order_by = tfast_sc_eldest_priority
				position = 0
				save_scope_as = tfast_sc_next_chosen
			}
		}
		tfast_sc_prophecy_promote_effect = { CANDIDATE = scope:tfast_sc_next_chosen }
	}
	else_if = {
		limit = {
			exists = global_var:tfast_sc_last_chosen
			global_var:tfast_sc_last_chosen = {
				any_child = {
					any_child = {
						is_alive = yes
						tfast_sc_in_designated_dynasty_trigger = yes
					}
				}
			}
		}
		global_var:tfast_sc_last_chosen = {
			ordered_child = {
				order_by = tfast_sc_eldest_priority
				position = 0
				ordered_child = {
					limit = {
						is_alive = yes
						tfast_sc_in_designated_dynasty_trigger = yes
					}
					order_by = tfast_sc_eldest_priority
					position = 0
					save_scope_as = tfast_sc_next_chosen
				}
			}
		}
		tfast_sc_prophecy_promote_effect = { CANDIDATE = scope:tfast_sc_next_chosen }
	}
	else_if = {
		limit = {
			exists = global_var:tfast_sc_last_chosen
			global_var:tfast_sc_last_chosen = {
				any_child = {
					any_child = {
						any_child = {
							is_alive = yes
							tfast_sc_in_designated_dynasty_trigger = yes
						}
					}
				}
			}
		}
		global_var:tfast_sc_last_chosen = {
			ordered_child = {
				order_by = tfast_sc_eldest_priority
				position = 0
				ordered_child = {
					order_by = tfast_sc_eldest_priority
					position = 0
					ordered_child = {
						limit = {
							is_alive = yes
							tfast_sc_in_designated_dynasty_trigger = yes
						}
						order_by = tfast_sc_eldest_priority
						position = 0
						save_scope_as = tfast_sc_next_chosen
					}
				}
			}
		}
		tfast_sc_prophecy_promote_effect = { CANDIDATE = scope:tfast_sc_next_chosen }
	}
	else = {
		# No living descendant at any of three depths: the bloodline is
		# extinct and the prophecy ends. Explicit, reported, final.
		tfast_sc_prophecy_end_effect = yes
	}
}


# ---------------------------------------------------------------------------
#  END OF THE PROPHECY
#  Root = the prophecy story.
# ---------------------------------------------------------------------------
tfast_sc_prophecy_end_effect = {
	set_global_variable = { name = tfast_sc_prophecy_ended value = yes }
	remove_global_variable = tfast_sc_prophecy_open
	remove_global_variable = tfast_sc_chosen_character
	every_player = {
		trigger_event = { id = tfast_sc.0404 days = { 20 60 } }
	}
	end_story = yes
	debug_log = "TFAST Shattered Crowns: the chosen bloodline is extinct; the prophecy has ended"
}


# ---------------------------------------------------------------------------
#  AWAKENING
#  Root = the chosen. Called by the prophecy story when the time is right
#  (age, or danger - see the story cycle's effect group).
# ---------------------------------------------------------------------------
tfast_sc_prophecy_awaken_effect = {
	if = {
		limit = {
			is_alive = yes
			NOT = { has_trait = tfast_sc_chosen_awakened }
		}

		add_trait = tfast_sc_chosen_awakened
		if = {
			limit = { NOT = { has_character_modifier = tfast_sc_chosen_power_1_modifier } }
			add_character_modifier = { modifier = tfast_sc_chosen_power_1_modifier }
		}
		set_variable = { name = tfast_sc_power_tier value = 1 }
		remove_character_flag = tfast_sc_chosen_hidden

		every_player = {
			trigger_event = { id = tfast_sc.0403 days = { 10 40 } }
		}
		if = {
			limit = { is_ai = no }
			send_interface_toast = {
				type = event_toast_effect_good
				title = tfast_sc_awakened_toast.t
				left_icon = root
			}
		}

		debug_log = "TFAST Shattered Crowns: THE CHOSEN EMPEROR HAS AWAKENED"
		debug_log_scopes = yes
	}
}


# ---------------------------------------------------------------------------
#  DEEPENING THE POWER (upgradeable chosen power, tier 1 -> 3)
#  Root = the chosen.
# ---------------------------------------------------------------------------
tfast_sc_prophecy_power_up_effect = {
	if = {
		limit = { var:tfast_sc_power_tier ?= 1 }
		remove_character_modifier = tfast_sc_chosen_power_1_modifier
		add_character_modifier = { modifier = tfast_sc_chosen_power_2_modifier }
		set_variable = { name = tfast_sc_power_tier value = 2 }
		custom_tooltip = tfast_sc_power_2_tt
	}
	else_if = {
		limit = { var:tfast_sc_power_tier ?= 2 }
		remove_character_modifier = tfast_sc_chosen_power_2_modifier
		add_character_modifier = { modifier = tfast_sc_chosen_power_3_modifier }
		set_variable = { name = tfast_sc_power_tier value = 3 }
		custom_tooltip = tfast_sc_power_3_tt
	}
}


# ---------------------------------------------------------------------------
#  TRUE DEATH
#  The only way a Hollow immortal can be permanently destroyed. Called by
#  the confrontation event after its outcome roll.
#  Param: TARGET (a Hollow immortal)
#
#  Overlord exclusion is enforced HERE, at the only place that kills:
#  an Overlord is never destroyed, only sealed (tfast_sc_seal_overlord_effect).
# ---------------------------------------------------------------------------
tfast_sc_true_death_effect = {
	if = {
		limit = { $TARGET$ = { has_character_flag = tfast_sc_overlord } }
		# Cannot permanently destroy the Overlord - seal him instead.
		tfast_sc_seal_overlord_effect = yes
	}
	else_if = {
		limit = { $TARGET$ = { is_alive = yes } }
		$TARGET$ = {
			set_variable = { name = tfast_sc_true_death value = yes }
			death = natural
		}
		debug_log = "TFAST Shattered Crowns: TRUE DEATH - a Hollow immortal has been permanently destroyed"
		debug_log_scopes = yes
	}
}


# ---------------------------------------------------------------------------
#  UNAWAKENED FALLBACK
#  If the chosen dies before ever awakening, the story passes on exactly
#  like any other succession; if the window closes with no chosen at all,
#  nothing happens - the prophecy simply never fired.
# ---------------------------------------------------------------------------
tfast_sc_prophecy_check_effect = {
	# Story-scope helper used by the effect groups.
	if = {
		limit = {
			NOT = {
				exists = global_var:tfast_sc_chosen_character
				global_var:tfast_sc_chosen_character = { is_alive = yes }
			}
		}
		tfast_sc_prophecy_succession_effect = yes
	}
	else_if = {
		limit = {
			global_var:tfast_sc_chosen_character = { is_alive = yes }
			NOT = { story_owner = global_var:tfast_sc_chosen_character }
		}
		make_story_owner = global_var:tfast_sc_chosen_character
	}
}
