namespace = tfast_sct

# Newborn Targaryen: a quarter of them get a dragon's egg in the cradle.
tfast_sct.0001 = {
	type = character_event
	hidden = yes

	trigger = {
		is_alive = yes
		has_character_flag = tfast_sc_targaryen
		NOT = { has_character_modifier = tfast_sct_dragon_egg_modifier }
	}

	immediate = {
		random = {
			chance = 25
			add_character_modifier = { modifier = tfast_sct_dragon_egg_modifier }
			trigger_event = { id = tfast_sct.0002 years = 5 }
		}
	}
}

# The egg hatches.
tfast_sct.0002 = {
	type = character_event
	theme = dynasty
	title = tfast_sct.0002.t
	desc = tfast_sct.0002.desc
	left_portrait = {
		character = root
		animation = personality_bold
	}

	trigger = {
		is_alive = yes
		has_character_modifier = tfast_sct_dragon_egg_modifier
	}

	immediate = {
		remove_character_modifier = tfast_sct_dragon_egg_modifier
		add_character_modifier = { modifier = tfast_sct_dragonbound_modifier }
	}

	option = {
		name = tfast_sct.0002.a
		add_prestige = 100
	}
}

# The Hollow gather. Fired yearly to a human Targaryen when the Host is ready.
tfast_sct.0010 = {
	type = character_event
	theme = war
	title = tfast_sct.0010.t
	desc = tfast_sct.0010.desc
	left_portrait = {
		character = root
		animation = worry
	}

	trigger = {
		is_ai = no
		is_alive = yes
		has_character_flag = tfast_sc_targaryen
		exists = global_var:tfast_sc_hollow_readiness
		global_var:tfast_sc_hollow_readiness >= 60
		NOT = { has_character_flag = tfast_sct_warned }
	}

	immediate = {
		add_character_flag = { flag = tfast_sct_warned years = 5 }
	}

	option = {
		name = tfast_sct.0010.a
		add_stress = 5
	}

	option = {
		name = tfast_sct.0010.b
		add_prestige = 50
	}
}
