# ---------------------------------------------------------------------------
#  THE RECKONING ENDS
#  With relief decisions: the refugee trains and rebuilding work that
#  follows every Reckoning, per the living-world design.
# ---------------------------------------------------------------------------
tfast_sc.0202 = {
	type = character_event
	theme = realm

	title = tfast_sc.0202.t
	desc = { desc = tfast_sc.0202.desc }

	option = {
		name = tfast_sc.0202.a
		# Open the gates to the refugee trains.
		ai_chance = { base = 75 }

		add_prestige = 200
		create_character = {
			template = local_noble_character
			employer = root
			gender_female_chance = 50
			save_scope_as = tfast_sc_refugee
		}
		hidden_effect = {
			# A life begun again: a claim on what the Hollow stripped.
			random_sub_realm_county = {
				limit = { has_county_modifier = tfast_sc_raided_modifier }
				save_temporary_scope_as = tfast_sc_ruined_county
			}
			if = {
				limit = { exists = scope:tfast_sc_ruined_county }
				scope:tfast_sc_refugee = {
					add_pressed_claim = scope:tfast_sc_ruined_county
				}
			}
		}
	}

	option = {
		name = tfast_sc.0202.b
		# The gates stay shut.
		ai_chance = { base = 25 }

		add_character_modifier = {
			modifier = tfast_sc_host_retreat_modifier
			years = 3
		}
		stress_impact = {
			base = medium_stress_impact_gain
		}
	}
}
