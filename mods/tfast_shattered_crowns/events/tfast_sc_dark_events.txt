# ---------------------------------------------------------------------------
#  THE HOLLOW TITHE
# ---------------------------------------------------------------------------
tfast_sc.0900 = {
	type = character_event
	theme = dread
	override_background = { reference = tfast_sc_hollow_omens }

	title = tfast_sc.0900.t
	desc = { desc = tfast_sc.0900.desc }

	option = {
		name = tfast_sc.0900.a
		# Pay what they ask.
		ai_chance = { base = 60 }

		remove_short_term_gold = 150
		set_variable = { name = tfast_sc_tithe_recent value = yes years = 4 }
		add_prestige = -50
	}

	option = {
		name = tfast_sc.0900.b
		# Refuse them.
		ai_chance = { base = 40 }

		capital_county = {
			add_county_modifier = {
				modifier = tfast_sc_raided_modifier
				years = 5
			}
		}
		tfast_sc_add_dark_resistance_effect = { AMOUNT = 1 }
		set_variable = { name = tfast_sc_tithe_recent value = yes years = 3 }
		add_prestige = 100
		stress_impact = {
			base = medium_stress_impact_gain
		}
	}
}


# ---------------------------------------------------------------------------
#  THE FEVER YEAR
# ---------------------------------------------------------------------------
tfast_sc.0901 = {
	type = character_event
	theme = dread

	title = tfast_sc.0901.t
	desc = { desc = tfast_sc.0901.desc }

	option = {
		name = tfast_sc.0901.a
		# Tend the sick yourself.
		ai_chance = { base = 30 }

		random = {
			chance = 30
			add_character_modifier = {
				modifier = tfast_sc_dark_malady_modifier
				years = 5
			}
		}
		random = {
			chance = 70
			add_prestige = 75
		}
		set_variable = { name = tfast_sc_fever_recent value = yes years = 4 }
	}

	option = {
		name = tfast_sc.0901.b
		# Seal the gates.
		ai_chance = { base = 40 }

		capital_county = {
			change_development_level = -1
		}
		set_variable = { name = tfast_sc_fever_recent value = yes years = 5 }
	}

	option = {
		name = tfast_sc.0901.c
		# Let it burn through.
		ai_chance = { base = 30 }

		random_courtier = {
			limit = { is_alive = yes }
			hidden_effect = { death = { death_reason = death_bubonic_plague } }
		}
		random_courtier = {
			limit = { is_alive = yes }
			hidden_effect = { death = { death_reason = death_bubonic_plague } }
		}
		capital_county = {
			add_county_modifier = {
				modifier = tfast_sc_dark_famine_modifier
				years = 4
			}
		}
		set_variable = { name = tfast_sc_fever_recent value = yes years = 5 }
		stress_impact = {
			base = medium_stress_impact_gain
		}
	}
}
