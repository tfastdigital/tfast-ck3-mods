#  tfast_sc.0501 -- news: an immortal has been destroyed

namespace = tfast_sc


# ---------------------------------------------------------------------------
#  THE TRUE DEATH CONFRONTATION
#  Fires on the awakened chosen. Target priority: one of the Ten, then any
#  Hollow immortal in the enemy court, then the Overlord himself (who can
#  only ever be sealed).
# ---------------------------------------------------------------------------
tfast_sc.0500 = {
	type = character_event
	theme = dynasty
	override_background = { reference = tfast_sc_reckoning }

	title = tfast_sc.0500.t
	desc = { desc = tfast_sc.0500.desc }

	immediate = {
		hidden_effect = {
			if = {
				limit = { tfast_sc_overlord_valid_trigger = yes }
				global_var:tfast_sc_overlord = { save_scope_as = tfast_sc_confront_overlord }

				if = {
					limit = {
						scope:tfast_sc_confront_overlord = {
							any_child = {
								is_alive = yes
								has_character_flag = tfast_sc_original_child
							}
						}
					}
					scope:tfast_sc_confront_overlord = {
						ordered_child = {
							limit = {
								is_alive = yes
								has_character_flag = tfast_sc_original_child
							}
							order_by = tfast_sc_eldest_priority
							position = 0
							save_scope_as = tfast_sc_true_death_target
						}
					}
				}
				else_if = {
					limit = {
						scope:tfast_sc_confront_overlord = {
							any_courtier_or_guest = {
								is_alive = yes
								has_trait = tfast_sc_hollow_immortal
							}
						}
					}
					scope:tfast_sc_confront_overlord = {
						random_courtier_or_guest = {
							limit = {
								is_alive = yes
								has_trait = tfast_sc_hollow_immortal
							}
							save_scope_as = tfast_sc_true_death_target
						}
					}
				}
				else = {
					scope:tfast_sc_confront_overlord = { save_scope_as = tfast_sc_true_death_target }
				}
			}
		}
	}

	option = {
		name = tfast_sc.0500.a

		trigger = { exists = scope:tfast_sc_true_death_target }

		ai_chance = { base = 100 }

		# The strike. Chance comes from tfast_sc_true_death_chance
		# (power tier, prowess, their defenses).
		random = {
			chance = tfast_sc_true_death_chance
			set_variable = { name = tfast_sc_td_result value = 1 }
		}

		if = {
			limit = { var:tfast_sc_td_result ?= 1 }
			remove_variable = tfast_sc_td_result
			custom_tooltip = tfast_sc_true_death_success_tt
			tfast_sc_record_memory_effect = { MEMORY = tfast_sc_memory_true_death YEARS = 150 }
			tfast_sc_true_death_effect = { TARGET = scope:tfast_sc_true_death_target }
			if = {
				limit = {
					NOT = {
						scope:tfast_sc_true_death_target = { has_character_flag = tfast_sc_overlord }
					}
				}
				every_player = {
					trigger_event = { id = tfast_sc.0501 days = { 10 60 } }
				}
			}
		}
		else = {
			remove_variable = tfast_sc_td_result
			custom_tooltip = tfast_sc_true_death_failure_tt
			add_stress = 30
			random = {
				chance = 30
				add_trait = wounded_1
			}
		}
	}

	option = {
		name = tfast_sc.0500.b

		ai_chance = { base = 0 }
	}
}

# ---------------------------------------------------------------------------
#  HIDDEN: A HOLLOW-BORN CHILD COMES INTO THEIR IMMORTALITY
#  The immortal trait freezes visual age, so it is granted at adulthood,
#  never at birth.
# ---------------------------------------------------------------------------
tfast_sc.0520 = {
	type = character_event
	hidden = yes

	trigger = {
		is_alive = yes
		has_character_flag = tfast_sc_hollow_born
	}

	immediate = {
		remove_character_flag = tfast_sc_hollow_born
		if = { limit = { NOT = { has_trait = tfast_sc_hollow_immortal } } add_trait = tfast_sc_hollow_immortal }
		set_immortal_age = 18
		if = { limit = { NOT = { has_character_flag = tfast_sc_hollow_host } } add_character_flag = tfast_sc_hollow_host }
	}
}
