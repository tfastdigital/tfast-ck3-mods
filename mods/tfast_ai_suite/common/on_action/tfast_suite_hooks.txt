# Realm automation. Buildings once a year. Council, court, and legacies
# every season, so an empty seat is not left until the next January.
yearly_playable_pulse = {
	on_actions = { tfast_suite_yearly }
}

quarterly_playable_pulse = {
	on_actions = { tfast_suite_quarterly }
}

tfast_suite_yearly = {
	effect = {
		if = {
			limit = {
				is_ai = no
				is_landed = yes
				has_character_flag = tfast_auto_build
			}
			tfast_auto_build_effect = yes
		}
	}
}

tfast_suite_quarterly = {
	effect = {
		if = {
			limit = {
				is_ai = no
				is_landed = yes
			}
			if = {
				limit = { has_character_flag = tfast_auto_council }
				tfast_auto_council_effect = yes
			}
			if = {
				limit = { has_character_flag = tfast_auto_court }
				tfast_auto_court_effect = yes
			}
			if = {
				limit = { has_character_flag = tfast_auto_legacy }
				tfast_auto_legacy_effect = yes
			}
		}
	}
}
