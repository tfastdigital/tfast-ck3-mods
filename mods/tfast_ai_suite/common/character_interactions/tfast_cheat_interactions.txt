# Right-click a character. Open TFAST. The list works like the debug interaction menu.
# Gold and the other currencies are amount buttons, not a single decision.

tfast_cheat_open = {
	category = interaction_category_tfast
	desc = tfast_cheat_open_desc
	common_interaction = yes
	icon = icon_gold
	interface_priority = 100
	auto_accept = yes
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes

	is_shown = {
		scope:actor = {
			is_ai = no
			NOT = { has_character_flag = tfast_cheat_open }
		}
	}

	on_send = {
		scope:actor = { add_character_flag = tfast_cheat_open }
	}
}

	}
}

	category = interaction_category_tfast
	desc = tfast_cheat_impregnate_desc
	common_interaction = yes

	interface_priority = 42
	auto_accept = yes
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes

	is_shown = {
		scope:actor = { is_ai = no has_character_flag = tfast_cheat_open }
		NOT = { scope:recipient = scope:actor }
		OR = {
			AND = {
				scope:actor = { is_male = yes }
				scope:recipient = { is_female = yes }
			}
			AND = {
				scope:actor = { is_female = yes }
				scope:recipient = { is_male = yes }
			}
		}
	}

	on_send = {
		if = {
			limit = { scope:recipient = { is_female = yes } }
			scope:recipient = { make_pregnant = { father = scope:actor } }
		}
		else = {
			scope:actor = { make_pregnant = { father = scope:recipient } }
		}
	}
}
