# Right-click another character. The player is the only one who can see these.
# They accept themselves, so nothing waits on the other person.

tfast_grant_gold_interaction = {
	category = interaction_category_friendly
	icon = icon_gold
	desc = tfast_grant_gold_interaction_desc
	auto_accept = yes

	is_shown = {
		scope:actor = { is_ai = no }
		scope:recipient = {
			is_alive = yes
			this != scope:actor
		}
	}

	on_accept = {
		scope:recipient = {
			add_gold = 1000
			add_opinion = {
				target = scope:actor
				modifier = grateful_opinion
				opinion = 20
			}
		}
	}
}
