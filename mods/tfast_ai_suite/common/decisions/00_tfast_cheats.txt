# ============================================================================
#  TFAST AI Suite -- Cheat decisions                             CK3 1.20.x
# ============================================================================
#  Design guarantees (why this cannot break a game)
#  ------------------------------------------------
#   * is_ai = no             -> only the human player can ever see these.
#   * ai_check_interval = 0  -> the AI never evaluates them, so the AI
#                               economy/warfare is completely untouched.
#   * Additive only          -> no vanilla file is overwritten. Disabling
#                               the mod leaves the game identical.
#   * Every effect used is a vanilla effect verified against 1.20.0.3.
#
#  They appear in the Decisions tab under "Major Decisions".
#
#  NOTE: this is a single-player convenience mod. Using it (like the in-game
#  debug console) disables achievements for that save.
# ============================================================================


# ------------------------------------------------------------------- GOLD ---
tfast_cheat_gold = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_spend_money.dds" }
	decision_group_type = major
	sort_order = 9000
	ai_check_interval = 0

	is_shown = { always = no is_ai = no
		has_character_flag = tfast_menu_open
		has_character_flag = tfast_menu_page_resources
	}
	is_valid = { is_ai = no }

	effect = {
		tfast_debug_note_effect = { NOTE = tfast_cheat_gold }
		add_gold = 10000
	}
}

# -------------------------------------------------------------- GOLDEN AGE --
# Everything at once, plus a strong but temporary realm-wide boon.
tfast_cheat_golden_age = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_golden_age.dds" }
	decision_group_type = major
	sort_order = 9012
	ai_check_interval = 0

	is_shown = { always = no is_ai = no
		has_character_flag = tfast_menu_open
		has_character_flag = tfast_menu_page_resources
	}
	is_valid = { is_ai = no }

	effect = {
		tfast_debug_note_effect = { NOTE = tfast_cheat_golden_age }
		custom_tooltip = tfast_cheat_golden_age_tt
		add_gold = 10000
		add_prestige = 10000
		add_piety = 10000
		if = {
			limit = { exists = dynasty }
			dynasty = { add_dynasty_prestige = 10000 }
		}
		else = { add_prestige = 10000 }
		add_legitimacy = 1000
		add_dread = 150
		add_tyranny = -1000
		add_stress = -1000
		if = {
			limit = { is_landed = yes }
			every_sub_realm_county = { change_development_level = 2 }
		}
		every_maa_regiment = { change_maa_regiment_size = 3 }
		add_character_modifier = {
			modifier = tfast_golden_age_modifier
			years = 20
		}
	}
}
