# ============================================================================
#  TFAST AI Suite -- realm automation decisions                 CK3 1.20.x
# ============================================================================
#  Open the cheat menu, then Automation. The switches keep working after
#  the menu is closed. Building spends tfast_build_budget gold per year
#  and always leaves 200 in the treasury.
# ============================================================================


tfast_menu_page_auto_open = {
	picture = { reference = "gfx/interface/illustrations/decisions/tfast_menu_tab_page.dds" }
	decision_group_type = major
	sort_order = 8998
	ai_check_interval = 0

	is_shown = {
		always = no
		is_ai = no
		has_character_flag = tfast_menu_open
		NOT = { tfast_menu_any_page_open_trigger = yes }
	}
	is_valid = { is_ai = no }

	effect = {
		tfast_menu_clear_pages_effect = yes
		add_character_flag = { flag = tfast_menu_page_auto }
	}
}

tfast_auto_realm_on = {
	picture = { reference = "gfx/interface/illustrations/decisions/tfast_menu_tab_page.dds" }
	decision_group_type = major
	sort_order = 9060
	ai_check_interval = 0

	is_shown = {
		always = no
		is_ai = no
		has_character_flag = tfast_menu_open
		has_character_flag = tfast_menu_page_auto
		NOT = { has_character_flag = tfast_auto_build }
	}
	is_valid = { is_landed = yes }

	effect = {
		tfast_debug_note_effect = { NOTE = tfast_auto_realm_on }
		add_character_flag = { flag = tfast_auto_build }
		add_character_flag = { flag = tfast_auto_council }
		add_character_flag = { flag = tfast_auto_court }
		add_character_flag = { flag = tfast_auto_legacy }
		if = {
			limit = { NOT = { has_variable = tfast_build_budget } }
			set_variable = { name = tfast_build_budget value = 500 }
		}
		tfast_auto_build_effect = yes
		tfast_auto_council_effect = yes
		tfast_auto_court_effect = yes
		tfast_auto_legacy_effect = yes
	}
}

	}
}
