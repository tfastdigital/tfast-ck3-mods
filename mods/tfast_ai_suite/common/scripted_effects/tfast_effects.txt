# Give away surplus titles until the holder is back inside their domain limit.
# Generates a local noble for each title and transfers it cleanly, using the
# same title-and-vassal-change ritual the vanilla "Grant Title" interaction
# uses, so opinion/vassal records stay consistent.
#
# The loop is BOUNDED on purpose. Vanilla bounds its own `while` blocks with a
# counter (see 00_decisions_effects.txt, `var:while_counter < 3`); a `while`
# with no ceiling is a free-hang waiting to happen if a grant ever fails to
# reduce `domain_limit_available`. The cap is far above any realistic domain
# overage, so in practice it is never reached.
#
# Parameters: none. Expects to run in a character scope.
tfast_grant_excess_titles_effect = {
	save_scope_as = tfast_giver
	set_variable = { name = tfast_grant_guard value = 0 }
	while = {
		limit = {
			domain_limit_available < 0
			var:tfast_grant_guard < 60
			any_held_title = {
				tfast_grantable_title_trigger = { GIVER = scope:tfast_giver }
			}
		}
		random_held_title = {
			limit = {
				tfast_grantable_title_trigger = { GIVER = scope:tfast_giver }
			}
			save_scope_as = tfast_title
			create_character = {
				template = local_noble_character
				location = scope:tfast_giver.capital_province
				save_scope_as = tfast_receiver
			}
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = tfast_change
				add_claim_on_loss = no
			}
			change_title_holder_include_vassals = {
				holder = scope:tfast_receiver
				change = scope:tfast_change
			}
			resolve_title_and_vassal_change = scope:tfast_change
		}
		change_variable = { name = tfast_grant_guard add = 1 }
	}
	remove_variable = tfast_grant_guard
}

# Close every cheat-menu page. The menu shell calls this when a page opens,
# when Back is clicked, and when the menu closes. Expects a character scope.
tfast_menu_clear_pages_effect = {
	remove_character_flag = tfast_menu_page_resources
	remove_character_flag = tfast_menu_page_domain
	remove_character_flag = tfast_menu_page_military
	remove_character_flag = tfast_menu_page_character
	remove_character_flag = tfast_menu_page_realm
	remove_character_flag = tfast_menu_page_auto
}

# Write one line to logs/debug.log when the player has turned TFAST debug on.
# Does nothing while debug is off, so a cheat stays a cheat.
# Parameter: NOTE = short name of the decision that just ran.
tfast_debug_note_effect = {
	if = {
		limit = { has_global_variable = tfast_debug }
		debug_log = "TFAST debug: $NOTE$"
		debug_log_scopes = yes
	}
}
